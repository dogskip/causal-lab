import { Simulation, MAX_EVENTS } from "./workspace/dist/simulation.js";
import { executeScenario, parseScenario } from "./workspace/dist/scenario.js";
import { createApp } from "./workspace/dist/app.js";
import { createReplay, frameAt } from "./workspace/web/frame.ts";
const findings=[];
const config={replicas:["a","b"],seed:1,minLatency:1,maxLatency:1,dropRate:0,duplicateRate:0};
const protoScenario=parseScenario({config:{...config,replicas:["__proto__","b"],dropRate:1},steps:[{at:0,action:"put",replica:"__proto__",key:"k",value:"left"},{at:0,action:"put",replica:"b",key:"k",value:"right"}]});
const protoReport=executeScenario(protoScenario);
const response=await createApp().request("/v1/run",{method:"POST",headers:{"content-type":"application/json"},body:JSON.stringify(protoScenario)});
findings.push({case:"reserved-replica-name",inputReplicas:protoScenario.config.replicas,reportedReplicas:Object.keys(protoReport.states),hiddenState:Object.getPrototypeOf(protoReport.states),httpStatus:response.status,httpReport:await response.json()});
const capped=new Simulation({...config,dropRate:1});
for(let i=0;i<MAX_EVENTS;i++) capped.put("a","k","accepted");
const before=capped.report();
let error;
try{capped.put("a","k","rejected");}catch(e){error=e.message;}
const after=capped.report();
findings.push({case:"rejected-operation-mutates",error,before:{states:before.states,versions:before.versions,traceLength:before.trace.length},after:{states:after.states,versions:after.versions,traceLength:after.trace.length}});
const replayConfig={...config,minLatency:5,maxLatency:5};
const replayScenario=parseScenario({config:replayConfig,steps:[{at:0,action:"put",replica:"a",key:"k",value:"v"},{at:20,action:"heal"}]});
const snapshots=[];
const replayReport=executeScenario(replayScenario,(step,states)=>snapshots.push({at:step.at,states}));
snapshots.push({at:replayReport.virtualTime,states:replayReport.states});
const frame=frameAt(createReplay(replayReport,snapshots),5);
const actual=new Simulation(replayConfig);actual.put("a","k","v");actual.advance(5);
findings.push({case:"replay-state-after-delivery",time:5,deliveries:frame.entries.filter(e=>e.kind==="delivered"),displayedState:frame.states,actualState:actual.states(),snapshotTimes:snapshots.map(s=>s.at)});
const futureScenario=parseScenario({config:replayConfig,steps:[{at:10,action:"put",replica:"a",key:"k",value:"future"}]});
const futureSnapshots=[];
const futureReport=executeScenario(futureScenario,(step,states)=>futureSnapshots.push({at:step.at,states}));
futureSnapshots.push({at:futureReport.virtualTime,states:futureReport.states});
findings.push({case:"replay-state-before-first-operation",time:0,displayedState:frameAt(createReplay(futureReport,futureSnapshots),0).states});
console.log(JSON.stringify(findings,null,2));
