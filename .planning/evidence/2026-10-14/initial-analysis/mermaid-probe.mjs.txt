import { chromium } from "/Users/bjchoi/.npm/_npx/705bc6b22212b352/node_modules/playwright/index.mjs";
import { traceToMermaid } from "./workspace/dist/trace.js";
import { writeFile } from "node:fs/promises";
const generated=traceToMermaid([{sequence:1,time:1,kind:"delivered",from:"a",to:"b",operation:"a:1"}]);
const browser=await chromium.launch({headless:true,executablePath:"/Applications/Google Chrome.app/Contents/MacOS/Google Chrome"});
try{
 const page=await browser.newPage();
 await page.setContent("<!doctype html><html><body></body></html>");
 const result=await page.evaluate(async code=>{
  const {default:mermaid}=await import("https://cdn.jsdelivr.net/npm/mermaid@11.12.3/dist/mermaid.esm.min.mjs");
  mermaid.initialize({startOnLoad:false,securityLevel:"strict"});
  const d=await mermaid.mermaidAPI.getDiagramFromText(code);
  const actors=d.db.getActors();
  return {actors:actors instanceof Map?Object.fromEntries(actors):actors,messages:d.db.getMessages()};
 },generated);
 const resultWithInput={generated,result};
 await writeFile(new URL("./mermaid-probe.json",import.meta.url),JSON.stringify(resultWithInput,null,2));
 console.log(JSON.stringify(resultWithInput,null,2));
}finally{await browser.close();}
